from Video

group by user_id

having count(DISTINCT date) = 365


Задача 2. Найти пользователей, смотревших видео 1 и 3, но не смотревших видео 2


select user_id
from Video
group by user_id
having sum(case when video_id = 1 then 1 else 0) >0 and 
sum(case when video_id = 3 then 1 else 0) >0 and 
sum(case when video_id = 2 then 1 else 0) = 0
